#include "../h/MemoryAllocator.hpp"
#include "../h/RISC_V.hpp"
#include "../h/syscall_c.hpp"
#include "../lib/hw.h"
#include "../test/printing.hpp"
#include "../h/TCB.hpp"

extern "C" void trap_handler ();

// --- niti koje se NIKAD ne završavaju (profesorov predlog za korak 6) ---
#define NUM_OF_ITERATIONS 100000000

static void worker ( void* arg ) {
    uint64 id = ( uint64 ) arg;
    uint64 i  = 0;
    while ( true ) {
        print_str ( "  nit " );    print_int ( id );
        print_str ( "  ciklus " ); print_int ( i ); print_str ( "\n" );
        i++;
        for ( volatile uint64 k = 0; k < NUM_OF_ITERATIONS; k++ ) { /* uspori ispis */ }
        thread_dispatch (); // dobrovoljno ustupi procesor
    }
}

void main () {

    // install trap_handler (stvec -> trap_handler)
    RISC_V::w_stvec ( ( uint64 ) &trap_handler );

    // 1) BOOTSTRAP glavne niti — predstavlja kontekst koji se SAD izvršava.
    //    body == nullptr  =>  bez steka, NE ide u scheduler, context se ne
    //    inicijalizuje (main je "stara" nit u prvom context_switch-u, pa mu
    //    se realni ra/sp tek tada upišu).
    TCB::running = new TCB ();

    // 2) kreiraj radne niti PREKO SISTEMSKOG POZIVA (ecall -> handler -> new TCB)
    thread_t t[3];
    thread_create ( &t[0], worker, ( void* ) 1 );
    thread_create ( &t[1], worker, ( void* ) 2 );
    thread_create ( &t[2], worker, ( void* ) 3 );

    // 3) glavna nit i sama učestvuje u rotaciji — takođe se ne završava
    while ( true ) {
        print_str ( "nit 0 (main)\n" );
        for ( volatile uint64 k = 0; k < NUM_OF_ITERATIONS; k++ ) {}
        thread_dispatch ();
    }

    print_str ( "Error: main" );

    *( ( uint32* ) 0x100000 ) = 0x5555; // halt the emulator
}
